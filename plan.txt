
Architectural Changes

• Authentication Upgrade: You must add the https://googleapis.com scope to your existing Google SSO configurations in Go.
• Token Storage: Your backend needs to store the user's Google access_token (and ideally a refresh_token) in their session or database.
• Streaming Endpoint: The PWA frontend will set the <audio> tag's src attribute directly to your Go backend (e.g., /api/stream?fileId=XXXX). The Go backend will fetch chunks from Google and pipe them straight to the user.

2. Go Backend Integration

You will need the official Google API libraries for Go. Run these commands in your project terminal:
bash
go get google.golang.org/api/drive/v3
go get golang.org/x/oauth2/google
Use code with caution.
Here is the robust backend implementation featuring an MP3 scanner endpoint and a high-performance, memory-efficient chunked streaming endpoint.
go
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"strings"

	"golang.org/x/oauth2"
	"google.golang.org/api/drive/v3"
	"google.golang.org/api/option"
)

// Helper function to reconstruct an OAuth2 client using the user's stored session token
func getDriveService(ctx context.Context, userAccessToken string) (*drive.Service, error) {
	token := &oauth2.Token{
		AccessToken: userAccessToken,
		TokenType:   "Bearer",
	}

	// Confining the client configuration directly to the token context
	client := oauth2.NewClient(ctx, oauth2.StaticTokenSource(token))

	srv, err := drive.NewService(ctx, option.WithHTTPClient(client))
	if err != nil {
		return nil, err
	}
	return srv, nil
}

// 1. Endpoint to scan and list MP3 files from the user's Google Drive
func ListMusicHandler(w http.ResponseWriter, r *http.Request) {
	ctx := context.Background()

	// TODO: Retrieve the active Google access token from your current SSO session mechanism
	userToken := r.Header.Get("X-User-Google-Token")
	if userToken == "" {
		http.Error(w, "Unauthorized: Missing Google token", http.StatusUnauthorized)
		return
	}

	srv, err := getDriveService(ctx, userToken)
	if err != nil {
		http.Error(w, "Failed to initialize Drive service", http.StatusInternalServerError)
		return
	}

	// Restrict query to audio/mpeg (MP3) files that are not in the trash bin
	query := "mimeType = 'audio/mpeg' and trashed = false"

	call := srv.Files.List().
		Q(query).
		Fields("files(id, name, size, mimeType)").
		PageSize(100)

	result, err := call.Do()
	if err != nil {
		http.Error(w, fmt.Sprintf("Drive API error: %v", err), http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(result.Files)
}

// 2. High-performance streaming proxy endpoint
func StreamMusicHandler(w http.ResponseWriter, r *http.Request) {
	ctx := context.Background()

	fileID := r.URL.Query().Get("fileId")
	userToken := r.URL.Query().Get("token") // Passed via URL query parameter for simple media tags

	if fileID == "" || userToken == "" {
		http.Error(w, "Missing fileId or token parameters", http.StatusBadRequest)
		return
	}

	srv, err := getDriveService(ctx, userToken)
	if err != nil {
		http.Error(w, "Failed to initialize Drive service", http.StatusInternalServerError)
		return
	}

	// 1. Fetch file metadata to get the proper size and name for HTTP range compatibility
	meta, err := srv.Files.Get(fileID).Fields("size, mimeType, name").Do()
	if err != nil {
		http.Error(w, "File metadata lookup failed", http.StatusNotFound)
		return
	}

	// 2. Execute a download request to fetch the raw file payload bytes
	downloadCall := srv.Files.Get(fileID)
	downloadCall.Header().Set("Accept", "application/octet-stream")

	// Forward incoming HTML5 audio byte-range requests directly up to Google Drive if present
	if rangeHeader := r.Header.Get("Range"); rangeHeader != "" {
		downloadCall.Header().Set("Range", rangeHeader)
	}

	response, err := downloadCall.Download()
	if err != nil {
		http.Error(w, fmt.Sprintf("Failed to fetch stream: %v", err), http.StatusInternalServerError)
		return
	}
	defer response.Body.Close()

	// 3. Mirror essential stream tracking headers back down to your frontend player
	w.Header().Set("Content-Type", meta.MimeType)
	if response.Header.Get("Content-Range") != "" {
		w.Header().Set("Content-Range", response.Header.Get("Content-Range"))
		w.WriteHeader(http.StatusPartialContent) // Return 206 status for successful scrubbing/skipping
	} else {
		w.Header().Set("Content-Length", fmt.Sprintf("%d", meta.Size))
	}

	w.Header().Set("Accept-Ranges", "bytes")

	// 4. Memory-efficient pipeline: Stream raw bytes straight out to client without server buffering
	_, err = io.Copy(w, response.Body)
	if err != nil {
		// Connection dropped early by user scrubbing or skipping the track
		return
	}
}

func main() {
	http.HandleFunc("/api/music/list", ListMusicHandler)
	http.HandleFunc("/api/music/stream", StreamMusicHandler)

	fmt.Println("Backend server running on :8080...")
	http.ListenAndServe(":8080", nil)
}
Use code with caution.

3. Simplified Frontend PWA Usage

Because your Go backend mirrors chunk headers (Accept-Ranges and Content-Range), the HTML5 <audio> player inside your PWA can naturally play, fast-forward, and scrub through tracks out-of-the-box without requiring complex JavaScript array buffers.
javascript
// 1. Fetch JSON track list from your backend
async function loadDriveSongs() {
  const res = await fetch('/api/music/list', {
    headers: { 'X-User-Google-Token': userAccessToken }
  });
  const songs = await res.json();

  // Render songs list
  songs.forEach(song => {
    const btn = document.createElement('button');
    btn.innerText = song.name;

    // 2. Point player source directly to Go streaming proxy route
    btn.onclick = () => {
      const player = document.getElementById('audio-player');
      player.src = `/api/music/stream?fileId=${song.id}&token=${encodeURIComponent(userAccessToken)}`;
      player.play();
    };
    document.body.appendChild(btn);
  });
}
